### Implementation Guidelines

#### 登录流程设计
```
1. 用户提交用户名/密码
2. 验证用户名/密码
3. 检查用户2FA状态
   - 如果未启用2FA：直接完成登录
   - 如果启用2FA：返回需要2FA验证的响应
4. 用户提交2FA验证码
5. 验证2FA代码
6. 完成登录，生成包含2FA状态的会话
```

#### API响应格式
登录API需要返回2FA状态信息：
```json
{
  "code": 200,
  "message": "success",
  "data": {
    "requiresTwoFactor": true,
    "tempToken": "临时token用于2FA验证"
  }
}
```

2FA验证API响应：
```json
{
  "code": 200,
  "message": "success",
  "data": {
    "token": "完整的JWT token",
    "user": {
      // 用户信息
    }
  }
}
```

#### 安全考虑
1. **时间窗口容错**：支持当前和前一个30秒窗口的TOTP代码
2. **备用码使用**：每个备用码只能使用一次
3. **失败限制**：连续失败次数限制，防止暴力破解
4. **会话安全**：确保2FA验证状态正确传递到会话中

```markdown
# 双因素认证设置指南

## 什么是双因素认证？

双因素认证（2FA）是一种额外的安全措施，除了用户名和密码外，还需要第二个验证因素来确认您的身份。这大大提高了账户的安全性。

## 启用双因素认证

### 步骤1：进入安全设置
1. 登录您的账户
2. 点击右上角的用户头像
3. 选择"个人设置"
4. 点击"安全"标签页

### 步骤2：启用2FA
1. 在"双因素认证"部分，点击"启用2FA"
2. 系统将显示一个二维码
```

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { LoginCredentials, TwoFactorVerification, User } from '@/types/auth'

export const useAuthStore = defineStore('auth', () => {
  // 状态
  const token = ref<string | null>(localStorage.getItem('token'))
  const tempToken = ref<string | null>(null)
  const user = ref<User | null>(null)
  const requiresTwoFactor = ref(false)
  const loginStep = ref<'credentials' | 'two-factor' | 'success'>('credentials')
  
  // 计算属性
  const isAuthenticated = computed(() => {
    return !!token.value && !!user.value && !requiresTwoFactor.value
  })
  
  const isPartiallyAuthenticated = computed(() => {
    return !!tempToken.value && requiresTwoFactor.value
  })
  
  // 操作
  const login = async (credentials: LoginCredentials) => {
    try {
      const response = await authApi.login(credentials)
      
      if (response.requiresTwoFactor) {
        tempToken.value = response.tempToken
        requiresTwoFactor.value = true
        loginStep.value = 'two-factor'
      } else {
        token.value = response.token
        user.value = response.user
        localStorage.setItem('token', response.token)
        loginStep.value = 'success'
      }
      
      return response
    } catch (error) {
      throw error
    }
  }
  
  const verifyTwoFactor = async (verification: TwoFactorVerification) => {
    try {
      const response = await authApi.verifyTwoFactor({
        ...verification,
        tempToken: tempToken.value
      })
      
      token.value = response.token
      user.value = response.user
      localStorage.setItem('token', response.token)
      
      // 清理临时状态
      tempToken.value = null
      requiresTwoFactor.value = false
      loginStep.value = 'success'
      
      return response
    } catch (error) {
      throw error
    }
  }
  
  const logout = async () => {
    try {
      if (token.value) {
        await authApi.logout()
      }
    } finally {
      // 清理所有状态
      token.value = null
      tempToken.value = null
      user.value = null
      requiresTwoFactor.value = false
      loginStep.value = 'credentials'
      localStorage.removeItem('token')
    }
  }
  
  const resetLoginState = () => {
    tempToken.value = null
    requiresTwoFactor.value = false
    loginStep.value = 'credentials'
  }
  
  return {
    // 状态
    token: readonly(token),
    tempToken: readonly(tempToken),
    user: readonly(user),
    requiresTwoFactor: readonly(requiresTwoFactor),
    loginStep: readonly(loginStep),
    
    // 计算属性
    isAuthenticated,
    isPartiallyAuthenticated,
    
    // 操作
    login,
    verifyTwoFactor,
    logout,
    resetLoginState
  }
})
```

```typescript
import type { Router } from 'vue-router'
import { useAuthStore } from '@/store/auth'
import { ElMessage } from 'element-plus'

export function setupRouterGuards(router: Router) {
  // 认证守卫
  router.beforeEach(async (to, from, next) => {
    const authStore = useAuthStore()
    
    // 公开路由，无需认证
    const publicRoutes = ['/login', '/register', '/forgot-password']
    if (publicRoutes.includes(to.path)) {
      // 如果已登录，重定向到首页
      if (authStore.isAuthenticated) {
        next('/dashboard')
        return
      }
      next()
      return
    }
    
    // 检查是否需要认证
    if (!authStore.isAuthenticated) {
      // 检查是否处于2FA验证状态
      if (authStore.isPartiallyAuthenticated && to.path === '/verify-2fa') {
        next()
        return
      }
      
      // 未认证，重定向到登录页
      next({
        path: '/login',
        query: { redirect: to.fullPath }
      })
      return
    }
    
    // 已认证，继续导航
    next()
  })
  
  // 2FA验证守卫
  router.beforeEach((to, from, next) => {
    const authStore = useAuthStore()
    
    // 如果处于2FA验证状态，只允许访问验证页面
    if (authStore.isPartiallyAuthenticated) {
      if (to.path !== '/verify-2fa') {
        next('/verify-2fa')
        return
      }
    }
    
    // 如果在2FA验证页面但不需要验证，重定向
    if (to.path === '/verify-2fa' && !authStore.isPartiallyAuthenticated) {
      next('/dashboard')
      return
    }
    
    next()
  })
  
  // 错误处理
  router.onError((error) => {
    console.error('路由错误:', error)
    ElMessage.error('页面加载失败，请重试')
  })
}
```

```typescript
import { ElMessage, ElNotification } from 'element-plus'
import type { AxiosError } from 'axios'

export interface ErrorContext {
  action: string
  component?: string
  userId?: string
}

export class ErrorHandler {
  static handle2FAError(error: AxiosError, context: ErrorContext) {
    const { response } = error
    
    if (!response) {
      this.handleNetworkError(context)
      return
    }
    
    switch (response.status) {
      case 400:
        this.handleValidationError(response.data, context)
        break
      case 401:
        this.handleAuthError(response.data, context)
        break
      case 429:
        this.handleRateLimitError(response.data, context)
        break
      case 500:
        this.handleServerError(context)
        break
      default:
        this.handleUnknownError(error, context)
    }
  }
  
  private static handleValidationError(data: any, context: ErrorContext) {
    const errorMessages = {
      'invalid_totp_code': '验证码无效，请检查并重新输入',
      'expired_totp_code': '验证码已过期，请使用最新的验证码',
      'invalid_backup_code': '备用码无效或已使用',
      'code_format_error': '请输入6位数字验证码'
    }
    
    const message = errorMessages[data.code] || data.message || '输入验证失败'
    
    ElMessage({
      type: 'error',
      message,
      duration: 5000
    })
  }
  
  private static handleAuthError(data: any, context: ErrorContext) {
    if (data.code === 'temp_token_expired') {
      ElNotification({
        title: '会话已过期',
        message: '请重新登录',
        type: 'warning',
        duration: 0
      })
      
      // 重定向到登录页
      window.location.href = '/login'
      return
    }
    
    ElMessage({
      type: 'error',
      message: '认证失败，请重试',
      duration: 3000
    })
  }
  
  private static handleRateLimitError(data: any, context: ErrorContext) {
    const retryAfter = data.retryAfter || 60
    
    ElNotification({
      title: '操作过于频繁',
      message: `请在${retryAfter}秒后重试`,
      type: 'warning',
      duration: 0
    })
  }
  
  private static handleNetworkError(context: ErrorContext) {
    ElNotification({
      title: '网络连接失败',
      message: '请检查网络连接并重试',
      type: 'error',
      duration: 0,
      showClose: true
    })
  }
  
  private static handleServerError(context: ErrorContext) {
    ElNotification({
      title: '服务器错误',
      message: '系统暂时不可用，请稍后重试',
      type: 'error',
      duration: 0
    })
  }
}
```

```typescript
  const retry = async (fn: () => Promise<any>) => {
    isRetrying.value = true
    try {
      await fn()
      clearError()
    } catch (err) {
      handleError(err, 'retry')
    } finally {
      isRetrying.value = false
    }
  }
  
  return {
    error: readonly(error),
    isRetrying: readonly(isRetrying),
    handleError,
    clearError,
    retry
  }
}
```
